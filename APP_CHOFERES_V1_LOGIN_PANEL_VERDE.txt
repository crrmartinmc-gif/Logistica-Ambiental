<script>
const BACKEND_URL = 'https://script.google.com/macros/s/AKfycbyLa6dr_bfxG8UKJoTbA69q6nShN8r4fmdy862KzqW7fqxrCyKy-D5q7zA02BjPtzdvYw/exec';

const $ = id => document.getElementById(id);
let session = null;

function escapeHtml(v){
  return String(v ?? '').replace(/[&<>"']/g, c => ({
    '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#039;'
  }[c]));
}

async function postJson(payload){
  const r = await fetch(BACKEND_URL,{
    method:'POST',
    headers:{'Content-Type':'text/plain;charset=utf-8'},
    body:JSON.stringify(payload)
  });
  return await r.json();
}

async function getJson(action, params={}){
  const qs = new URLSearchParams({accion:action,...params});
  const r = await fetch(BACKEND_URL+'?'+qs.toString());
  return await r.json();
}

function showError(msg){
  $('loginError').textContent = msg;
  $('loginError').style.display = 'block';
}

function initials(name){
  const p=String(name||'Chofer').trim().split(/\s+/);
  return (p[0]?.[0]||'C')+(p[1]?.[0]||'');
}

async function cargarDashboard(){
  const idChofer=session.idChofer;
  try{
    const [aRes,vRes,sRes] = await Promise.all([
      getJson('asignacionActiva',{idChofer}),
      getJson('listarVehiculos'),
      getJson('listarServicios')
    ]);

    const asignacion=aRes.asignacion || null;
    const vehiculos=Array.isArray(vRes.vehiculos)?vRes.vehiculos:[];
    const servicios=Array.isArray(sRes.servicios)?sRes.servicios:[];

    const vehiculo=asignacion
      ? vehiculos.find(v=>String(v.idUnidad)===String(asignacion.idUnidad))
      : null;

    renderVehicle(asignacion,vehiculo);

    const hoy=new Date().toISOString().slice(0,10);
    const propios=servicios.filter(s =>
      String(s.idChofer||'')===String(idChofer) &&
      String(s.estado||'').toUpperCase()!=='FINALIZADO'
    );

    propios.sort((a,b)=>String(a.fecha||'').localeCompare(String(b.fecha||'')) ||
                         String(a.horaProgramada||'').localeCompare(String(b.horaProgramada||'')));

    renderService(propios[0]||null);
  }catch(err){
    $('serviceBox').innerHTML='<div class="empty">No se pudieron consultar los datos operativos.<br>Intentá actualizar en unos segundos.</div>';
    console.error(err);
  }
}

function renderVehicle(a,v){
  if(!a){
    $('vehicleName').textContent='Sin asignación activa';
    $('vehiclePlate').textContent='—';
    $('vehicleInfo').innerHTML='<div class="empty">No tenés un vehículo asignado actualmente.</div>';
    return;
  }
  $('vehicleName').textContent=(v?.marca||'Vehículo')+' '+(v?.modelo||'');
  $('vehiclePlate').textContent=v?.dominio || a.dominio || 'SIN DOMINIO';
  $('vehicleInfo').innerHTML=[
    ['Interno',v?.numeroInterno||a.numeroInterno||'—'],
    ['Unidad',a.idUnidad||'—'],
    ['Base',a.base||v?.base||'—'],
    ['Turno',a.turno||'—']
  ].map(x=>`<div class="info-row"><span>${escapeHtml(x[0])}</span><strong>${escapeHtml(x[1])}</strong></div>`).join('');
}

function renderService(s){
  if(!s){
    $('serviceBox').innerHTML='<div class="empty"><strong>No tenés servicios pendientes.</strong><br><span>Cuando administración te asigne uno, aparecerá acá.</span></div>';
    return;
  }
  $('serviceBox').innerHTML=`
    <div class="service-card">
      <div class="service-head">
        <div class="eyebrow" style="color:#bfead5">${escapeHtml(s.tipo||'SERVICIO')}</div>
        <h2>${escapeHtml(s.nombreServicio||s.descripcion||'Servicio')}</h2>
      </div>
      <div class="route">
        <div class="route-point"><div class="dot"></div><div><strong>${escapeHtml(s.destinoInicial||'Origen no informado')}</strong><small>Origen</small></div></div>
        <div class="arrow">↓</div>
        <div class="route-point"><div class="dot"></div><div><strong>${escapeHtml(s.destinoFinal||'Destino no informado')}</strong><small>Destino</small></div></div>
      </div>
      <div class="service-meta">
        <div class="meta"><span>Horario</span><strong>${escapeHtml(formatearHora(s.horaProgramada))}</strong></div>
        <div class="meta"><span>Estado</span><strong>${escapeHtml(s.estado||'PROGRAMADO')}</strong></div>
      </div>
      <div class="actions" style="padding:0 19px 19px">
        <button class="primary" type="button" onclick="verServicio('${escapeHtml(s.idServicio||'')}')">VER SERVICIO</button>
      </div>
    </div>`;
}

function formatearHora(v){
  const s=String(v??'').trim();
  if(!s) return '—';
  const m=s.match(/T(\\d{2}):(\\d{2})/);
  if(m) return m[1]+':'+m[2];
  const h=s.match(/(\\d{1,2}):(\\d{2})/);
  if(h) return h[1].padStart(2,'0')+':'+h[2];
  return s;
}

function verServicio(id){
  alert('Servicio '+id+'\\n\\nLa siguiente etapa será la pantalla de inspección y el flujo operativo.');
}

$('loginForm').addEventListener('submit',async e=>{
  e.preventDefault();
  $('loginError').style.display='none';
  $('loginBtn').disabled=true;
  $('loginLoading').style.display='block';

  try{
    const res=await postJson({
      accion:'loginChofer',
      usuario:$('usuario').value,
      contrasena:$('contrasena').value
    });

    const data=res?.resultado||res;
    if(!res?.ok || !data?.login) throw new Error(res?.error||data?.error||'Usuario o contraseña incorrectos.');

    session={
      idChofer:data.idChofer,
      nombre:data.nombre,
      usuario:data.usuario,
      estado:data.estado
    };

    sessionStorage.setItem('diaVerdeChofer',JSON.stringify(session));

    $('driverName').textContent=session.nombre;
    $('topUserName').textContent=session.nombre;
    $('avatar').textContent=initials(session.nombre);
    $('topUser').classList.remove('hidden');
    $('loginScreen').classList.add('hidden');
    $('dashboardScreen').classList.remove('hidden');

    await cargarDashboard();
  }catch(err){
    showError(err.message||'No se pudo iniciar sesión.');
  }finally{
    $('loginBtn').disabled=false;
    $('loginLoading').style.display='none';
  }
});

(function restore(){
  try{
    const raw=sessionStorage.getItem('diaVerdeChofer');
    if(!raw) return;
    session=JSON.parse(raw);
    if(!session?.idChofer) return;
    $('driverName').textContent=session.nombre||'Chofer';
    $('topUserName').textContent=session.nombre||'Chofer';
    $('avatar').textContent=initials(session.nombre);
    $('topUser').classList.remove('hidden');
    $('loginScreen').classList.add('hidden');
    $('dashboardScreen').classList.remove('hidden');
    cargarDashboard();
  }catch(e){}
})();
